rich

groq

faster-whisper    # local offline fallback (Groq Whisper is primary, via groq sdk)

# --- voice: capture / wake / vad / playback ---
numpy
sounddevice       # mic capture + audio playback
soundfile         # wav read/write
vosk              # free offline wake word — custom phrase "wake up jarvis" (default engine)
openwakeword      # alt wake engine ("hey_jarvis") when WAKE_ENGINE=oww
silero-vad        # end-of-speech + barge-in VAD
torch             # silero-vad backend + Phase 5 emotion models
transformers      # Phase 5: local emotion (distilroberta) + voice-tone (wav2vec2 SER)

# --- Phase 6: tools + agent ---
psutil            # system status
keyboard          # media keys / volume
pyautogui         # screen control
mss               # screenshots (Phase 2 vision: screen capture)
opencv-python-headless  # Phase 2 vision: webcam capture (no GUI deps)
yt-dlp            # resolve YouTube videos for play_youtube
win11toast        # real Windows toast notifications (timers/reminders)

# --- Phase 10.L: always-on (auto-start + headless background) ---
# The headless supervisor + Task-Scheduler autostart use only the stdlib (subprocess/schtasks),

pystray           # optional tray icon (status dot, mic-mute, open/restart/quit) — Windows tray
